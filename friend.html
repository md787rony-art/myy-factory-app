// অটোমেটিক সিরিয়াল/রেজিস্ট্রেশন নম্বর জেনারেট করার ফাংশন function generateSerialNo() { // লোকাল স্টোরেজ থেকে আগের কাউন্ট নেওয়া (যদি না থাকে তবে ১০০১ থেকে শুরু হবে) let lastSerial = localStorage.getItem('player_serial_count') || 1000; let nextSerial = parseInt(lastSerial) + 1; localStorage.setItem('player_serial_count', nextSerial); return "REG-2026-" + nextSerial; } function completeRegistration() { const card = document.getElementById('poster-card'); // ১. অটোমেটিক সিরিয়াল/রেজিস্ট্রেশন নম্বর তৈরি let autoSerialNo = generateSerialNo(); document.getElementById('val_reg').innerText = autoSerialNo; // কার্ডে নম্বর বসবে // ২. কার্ডটি ছবি হিসেবে ডাউনলোড html2canvas(card, { scale: 2 }).then(canvas => { let a = document.createElement('a'); a.download = `Player_Card_${autoSerialNo}.png`; a.href = canvas.toDataURL('image/png'); a.click(); }); // ৩. তথ্য সংগ্রহ let nameBn = document.getElementById('in_name_bn').value; let nameEn = document.getElementById('in_name_en').value; let waNum = document.getElementById('in_wa').value; let adminNumber = "8801843416438"; // ৪. হোয়াটসঅ্যাপ মেসেজ ফরম্যাট (সিরিয়াল নম্বর সহ) let textMsg = `*নতুন প্লেয়ার রেজিস্ট্রেশন সম্পন্ন*%0A` + `🆔 *সিরিয়াল/রেজিস্ট্রেশন নং:* ${autoSerialNo}%0A` + `👤 *নাম (বাংলা):* ${nameBn}%0A` + `🔤 *নাম (ইংরেজি):* ${nameEn}%0A` + `📱 *হোয়াটসঅ্যাপ:* ${waNum}%0A%0A` + `📎 _দয়া করে ডাউনলোড করা কার্ডের ছবিটি নিচে অ্যাটাচ করে সেন্ড করুন।_`; let waLink = `https://wa.me/${adminNumber}?text=${textMsg}`; // ১.৫ সেকেন্ড পর হোয়াটসঅ্যাপ খুলবে setTimeout(() => { window.open(waLink, '_blank'); }, 1500); }
