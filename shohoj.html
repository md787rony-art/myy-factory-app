// ৫. ছবি আপলোড ও কমপ্রেসন (ফিক্সড ভার্সন) function processAndUpload(event, index) { const file = event.target.files[0]; if (!file) return; let statusBox = document.getElementById(`status-${index}`); statusBox.style.color = "#f59e0b"; statusBox.innerText = "⏳ ছবি প্রসেস ও আপলোড হচ্ছে..."; const reader = new FileReader(); reader.onload = function(e) { const img = new Image(); img.onload = function() { // ছবির সাইজ ছোট করা const canvas = document.createElement('canvas'); const MAX_WIDTH = 500; let width = img.width; let height = img.height; if (width > MAX_WIDTH) { height = Math.round((height * MAX_WIDTH) / width); width = MAX_WIDTH; } canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0, width, height); // Canvas কে Base64 এ রূপান্তর const base64Image = canvas.toDataURL('image/jpeg', 0.8).split(',')[1]; // FormData তৈরি করে ImgBB-তে পাঠানো let formData = new FormData(); formData.append("key", IMGBB_API_KEY); formData.append("image", base64Image); fetch("https://api.imgbb.com/1/upload", { method: "POST", body: formData }) .then(res => res.json()) .then(result => { if (result.success && result.data && result.data.url) { let imageUrl = result.data.url; products[index].img = imageUrl; document.getElementById(`preview-${index}`).src = imageUrl; statusBox.style.color = "#10b981"; statusBox.innerText = "✅ ছবি আপলোড সম্পন্ন!"; } else { statusBox.style.color = "#ef4444"; statusBox.innerText = "❌ আপলোড ব্যর্থ! অন্য ছবি চেষ্টা করুন।"; } }) .catch(err => { statusBox.style.color = "#ef4444"; statusBox.innerText = "❌ নেটওয়ার্ক ত্রুটি!"; }); }; img.src = e.target.result; }; reader.readAsDataURL(file); }
